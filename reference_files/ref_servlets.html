    Simple HTTP Servlet - Hello World

    Objective: To create a simple HTTP servlet that displays "Hello"
    💻 Java Code:
    import java.io.IOException;
import java.io.PrintWriter;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

public class HelloServlet extends HttpServlet {
    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        
        // Set content type
        response.setContentType("text/html");
        
        // Print "Hello"
        PrintWriter out = response.getWriter();
        out.println("
    Hello
    ");
    }
}

    Pizza Order Form with Servlet

    Objective: To create a program that takes input from user for Pizza order and displays the output for the options user has chosen using servlet
    Pizza order form with various input options
    💻 OrderServlet1.java:
    import java.io.IOException;
import java.io.PrintWriter;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet("/order1")
public class OrderServlet1 extends HttpServlet {
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        response.setContentType("text/html");
        PrintWriter out = response.getWriter();
        
        // Read parameters from the request
        String name = request.getParameter("name");
        String crust = request.getParameter("crust");
        String[] toppings = request.getParameterValues("toppings");
        String appetizer = request.getParameter("appetizer");
        String address = request.getParameter("address");
        String cardType = request.getParameter("cardType");
        String cardNumber = request.getParameter("cardNumber");
        String confirmCardNumber = request.getParameter("confirmCardNumber");
        
        // Basic validation for card number match
        if (cardNumber == null || !cardNumber.equals(confirmCardNumber)) {
            out.println("
    Credit card numbers do not match. Please go back and try again.
    ");
            return;
        }
        
        // Start printing the order details in table format
        out.println("Your Order");
        out.println("
    Your Order...
    ");
        out.println("");
        
        // Print parameter name and value rows
        printRow(out, "submit", "Order Pizza");
        printRow(out, "address", address);
        printRow(out, "name", name);
        printRow(out, "appetizer", appetizer);
        printRow(out, "cardType", cardType);
        printRow(out, "crust", crust);
        
        // Print card number as bullet points
        out.print("");
        
        // Print toppings as list
        out.print("");
        out.println("
    cardNum
      ");
        for (int i = 0; i < cardNumber.length(); i += 5) {
            int end = Math.min(i + 5, cardNumber.length());
            out.print("
    • " + cardNumber.substring(i, end) + "
      • ");
        }
        out.println("
    toppings
      ");
        if (toppings != null) {
            for (String topping : toppings) {
                out.print("
    • " + topping + "
      • ");
            }
        }
        out.println("
    ");
        out.println("");
    }
    
    private void printRow(PrintWriter out, String paramName, String paramValue) {
        out.println("");
        out.println("" + paramName + "");
        out.println("" + paramValue + "");
        out.println("");
    }
}
    Pizza order confirmation with selected options

    Login Authentication Servlet

    Objective: To create a program that takes username and password from user, checks if they are correct, and prints welcome message or error message using servlets
    Login form for authentication
    💻 LoginServlet.java:
    import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

public class LoginServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;
    
    // Hardcoded username and password for demo
    private final String USERNAME = "admin";
    private final String PASSWORD = "password123";

    protected void doPost(HttpServletRequest request, HttpServletResponse response) 
            throws ServletException, IOException {
        
        // Get username and password from form
        String username = request.getParameter("username");
        String password = request.getParameter("password");
        
        response.setContentType("text/html");
        
        if (USERNAME.equals(username) && PASSWORD.equals(password)) {
            // Credentials are correct — print welcome message
            response.getWriter().println("
    Welcome, " + username + "!
    ");
        } else {
            // Credentials are incorrect — print error message
            response.getWriter().println("
    Invalid username or password. Please try again.
    ");
            response.getWriter().println("Go back to Login");
        }
    }
    
    // Optional: show the login form on GET request
    protected void doGet(HttpServletRequest request, HttpServletResponse response) 
            throws ServletException, IOException {
        response.setContentType("text/html");
        response.getWriter().println("
    ");
        response.getWriter().println("Username: 
    ");
        response.getWriter().println("Password: 
    ");
        response.getWriter().println("");
        response.getWriter().println("
    ");
    }
}
